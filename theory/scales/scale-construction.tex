\begin{luacode*}
local letters = {"C", "D", "E", "F", "G", "A", "B"}
local natural_pitches = {0, 2, 4, 5, 7, 9, 11}
local accidentals = {[-1] = "♭", [0] = "", [1] = "♯"}
local tonics = {C = {1, 0}, A = {6, 0}, Bflat = {7, -1}}

-- Semitones above the tonic, indexed by degree even when a degree is omitted.
local major = {0, 2, 4, 5, 7, 9, 11}
local natural_minor = {0, 2, 3, 5, 7, 8, 10}

local function omit_degrees(scale, first, second)
  local pentatonic = {table.unpack(scale)}
  pentatonic[first], pentatonic[second] = nil, nil
  return pentatonic
end

local scales = {
  major = major,
  natural_minor = natural_minor,
  major_pentatonic = omit_degrees(major, 4, 7),
  minor_pentatonic = omit_degrees(natural_minor, 2, 6),
}

local function scale_data(scale, tonic)
  local degrees, steps, notes = {}, {}, {}
  local tonic_letter, tonic_accidental = table.unpack(tonic)
  local previous = 0
  for degree, major_pitch in ipairs(major) do
    local pitch = scale[degree]
    if pitch then
      -- Choose the letter from the degree, then set its pitch with an accidental.
      local letter = (tonic_letter + degree - 2) % #letters + 1
      local natural_interval = (natural_pitches[letter] - natural_pitches[tonic_letter]) % 12
      local accidental = tonic_accidental + pitch - natural_interval
      notes[#notes + 1] = letters[letter] .. assert(accidentals[accidental])
      degrees[#degrees + 1] = accidentals[pitch - major_pitch] .. degree
      if degree > 1 then
        steps[#steps + 1] = pitch - previous
      end
      previous = pitch
    end
  end
  steps[#steps + 1] = 12 - previous
  notes[#notes + 1] = notes[1]
  return degrees, steps, notes
end

function PrintScaleColumns(row, scale, tonic)
  local degrees, steps, notes = scale_data(scales[scale], tonics[tonic])
  tex.sprint("\\ScaleColumns{" .. row .. "}{"
    .. table.concat(degrees, " · ") .. "}{"
    .. table.concat(steps, " · ") .. "}{"
    .. table.concat(notes, "\\enspace ") .. "}")
end

function PrintScaleNotes(scale, tonic)
  local _, _, notes = scale_data(scales[scale], tonics[tonic])
  tex.sprint(table.concat(notes, "\\enspace "))
end
\end{luacode*}

  \GuideSection{\Localize{Параллельные гаммы}{Relative scales}}{%
    \Localize
      {До мажор и натуральный ля минор содержат одни и те же семь нот.
       Их пентатоники содержат одни и те же пять нот. В обеих парах при

       а A становится тоникой вместо ступени 6.}
      {C major and A natural minor share seven notes, and their pentatonics
       share five. In each pair, as the tonic moves from C to A, C changes
       from degree 1 to ♭3 and A changes from degree 6 to 1.}}%
  \GuideSection{\Localize{Одноимённые гаммы}{Parallel scales}}{%
    \Localize
      {До мажор и натуральный до минор имеют общую тонику C. В натуральном
       миноре ступени 3, 6 и 7 понижены на полутон. Из E, A и B получаются
       E♭, A♭ и B♭.}
      {C major and C natural minor share the tonic C. Natural minor lowers
       degrees 3, 6, and 7 of C major by one semitone, changing E, A, and B
       to E♭, A♭, and B♭.}%
    \ExampleNotes{\Localize{Натуральный до минор}{C natural minor}}{natural_minor}{C}}%

\newcommand{\ConstructionGuide}{%
  \GuideSection{\Localize{Пять нот из семи}{Five notes from seven}}{%
    \Localize
      {Из мажора уберите ступени 4 и 7 (F и B в до мажоре). В натуральном
       миноре уберите вторую и шестую ноты (B и F в ля миноре).
       Шестая нота минора записывается в формуле как ♭6.}
      {From major, omit degrees 4 and 7 (F and B in C major). From natural
       minor, omit its second and sixth notes (B and F in A minor).
       The sixth note of natural minor is ♭6 in the major-based formula.}}%
  \GuideSection{\Localize{Транспонирование}{Transposition}}{%
    \Localize
      {Перенесём минорную пентатонику из A в B♭, сохранив формулу и шаги.
       В си-бемоль мажоре возьмите ступени 1, 3, 4, 5 и 7. Понизьте
       третью (D) и седьмую (A) на полутон до D♭ и A♭.}
      {Move A minor pentatonic to B♭, keeping the formula and step pattern.
       Take degrees 1, 3, 4, 5, and 7 of B♭ major. Lower the third (D) and
       seventh (A) one semitone to D♭ and A♭.}%
    \ExampleNotes{\Localize{Минорная пентатоника от B♭}{B♭ minor pentatonic}}{minor_pentatonic}{Bflat}%
    \Localize
      {Здесь D♭ - ступень ♭3. В равномерной темперации D♭ и C♯ звучат
       одинаково, но буква D сохраняет роль третьей ступени от B♭.}
      {Here D♭ is ♭3. In equal temperament D♭ and C♯ sound the same,
       but the letter D preserves its role as the third degree above B♭.}}%
}
