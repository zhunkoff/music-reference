\begin{luacode*}
-- One data model and calculation for both language versions.
local labels = {
  "C", "C♯/D♭", "D", "D♯/E♭", "E", "F",
  "F♯/G♭", "G", "G♯/A♭", "A", "A♯/B♭", "B"
}
local function frequency(octave, pitch)
  local n = 12 * (octave - 4) + pitch - 9
  return 440 * 2 ^ (n / 12)
end
local function display(value)
  local integer, fraction = string.format("%.2f", value):match("(%d+)%.(%d+)")
  -- At most five integer digits in C0-B10; separate thousands with a thin space.
  integer = integer:gsub("^(%d+)(%d%d%d)$", "%1\\,%2")
  return integer .. "\\ChartDecimal{}" .. fraction
end
-- MIDI indices are used only to identify endpoints, not to restrict the chart.
-- These definitions drive both the cell markers and the bilingual range legend.
local instruments = {
  {low=40, high=83, ru="Классическая гитара, 19 ладов",
   en="Classical guitar, 19 frets", source=6},
  {low=40, high=88, ru="Электрогитара, 24 лада",
   en="Electric guitar, 24 frets", source=7},
  {low=28, high=67, ru="Бас, 4 струны, 24 лада",
   en="Bass guitar, 4 strings, 24 frets", source=8},
  {low=28, high=67, ru="Контрабас, 4 струны",
   en="Double bass, 4 strings", suffix="*", source=9},
  {low=64, high=93, ru="Балалайка-прима, 24 лада",
   en="Prima balalaika, 24 frets", source=10}
}
local function note_name(midi)
  return labels[midi % 12 + 1] .. tostring(math.floor(midi / 12) - 1)
end
function DrawInstrumentLegend()
  for number, instrument in ipairs(instruments) do
    tex.sprint(string.format("\\InstrumentRow{%d}{%s}{%s}{%s}{%s}{%d}",
      number, instrument.ru, instrument.en, note_name(instrument.low),
      note_name(instrument.high) .. (instrument.suffix or ""), instrument.source))
  end
end
function DrawNoteFrequencies()
  for index, label in ipairs(labels) do
    local pitch = index - 1
    tex.sprint(string.format("\\NoteLabel{%d}{%s}", pitch, label))
    for octave = 0, 10 do
      local f = frequency(octave, pitch)
      local status = 0
      if f < 20 then status = -1 elseif f > 20000 then status = 1 end
      tex.sprint(string.format("\\FrequencyCell{%d}{%d}{%s}{%d}",
        octave, pitch, display(f), status))
      local midi = 12 * (octave + 1) + pitch
      local marks = {}
      for number, instrument in ipairs(instruments) do
        if midi == instrument.low or midi == instrument.high then
          marks[#marks + 1] = tostring(number)
        end
      end
      if #marks > 0 then
        tex.sprint(string.format("\\InstrumentMarks{%d}{%d}{%s}",
          octave, pitch, table.concat(marks, ",")))
      end
    end
  end
end
\end{luacode*}
