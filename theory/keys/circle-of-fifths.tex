\begin{luacode*}
local sharps = {"F♯", "C♯", "G♯", "D♯", "A♯", "E♯", "B♯"}
local flats = {"B♭", "E♭", "A♭", "D♭", "G♭", "C♭", "F♭"}

-- Signed fifths from C: negative for flats, positive for sharps.
-- Each entry spells the major tonic and its relative minor tonic.
local keys = {
  [-7] = {"C♭", "A♭"},
  [-6] = {"G♭", "E♭"},
  [-5] = {"D♭", "B♭"},
  [-4] = {"A♭", "F"},
  [-3] = {"E♭", "C"},
  [-2] = {"B♭", "G"},
  [-1] = {"F",  "D"},
  [0]  = {"C",  "A"},
  [1]  = {"G",  "E"},
  [2]  = {"D",  "B"},
  [3]  = {"A",  "F♯"},
  [4]  = {"E",  "C♯"},
  [5]  = {"B",  "G♯"},
  [6]  = {"F♯", "D♯"},
  [7]  = {"C♯", "A♯"},
}

local function emit(command, ...)
  tex.sprint("\\" .. command .. "{" .. table.concat({...}, "}{") .. "}")
end

function PrintFifthsCircle()
  for fifths = -#flats, #sharps do
    local position = fifths % 12
    local offset = 0
    local count = tostring(math.abs(fifths))
    if fifths < 0 then
      count = count .. "♭"
      if fifths + 12 <= #sharps then
        offset = -17 -- Lower row in a shared sector.
      end
    elseif fifths > 0 then
      count = count .. "♯"
      if fifths - 12 >= -#flats then
        offset = 17 -- Upper row in a shared sector.
        emit("SplitSector", position)
      end
    end
    emit("KeyPair", position, offset, keys[fifths][1], keys[fifths][2], count)
  end
end

function PrintSignatureTable()
  for count = 1, #sharps do
    emit("SignatureRow", count,
      table.concat(sharps, "\\ ", 1, count),
      table.concat(flats, "\\ ", 1, count))
  end
end
\end{luacode*}

\newcommand{\DrawGuide}{%
  \node[anchor=north west,inner sep=0pt,outer sep=0pt,text=ChartInk,
    text width=320bp,align=left,font=\fontsize{11bp}{14bp}\selectfont]
    at (\SidebarLeft,337) {%
    \GuideSection{\Localize{Движение по кругу}{Follow the circle}}{\Localize
      {По часовой стрелке шаг равен чистой квинте вверх, против часовой —
       чистой кварте вверх. От C по часовой (\mbox{C → G → D}) добавляются
       диезы, против часовой (\mbox{C → F → B♭}) — бемоли.}
      {Go up a perfect fifth clockwise. Go up a perfect fourth counterclockwise.
       The path \mbox{C → G → D} adds sharps, and \mbox{C → F → B♭} adds flats.}}%
    \GuideSection{\Localize{Параллельные и одноимённые}{Relative and parallel keys}}{\Localize
      {Читайте пару в одной строке сектора. У параллельных G мажора и
       E минора общий ключевой знак F♯. У одноимённых G мажора и G минора
       общая тоника. Знаки гармонического и мелодического минора пишут у нот.}
      {Read across one row. G major and E minor are relative keys and share F♯
       in the signature. G major and G minor are parallel keys and share
       a tonic. Harmonic and melodic minor add accidentals in the music.}}%
    \GuideSection{\Localize{Тоника, субдоминанта и доминанта}{Tonic, subdominant, and dominant}}{\Localize
      {Для D мажора это D (I), G (IV, шаг против часовой) и A
       (V, шаг по часовой). Квинта вверх от B даёт F♯, от C♭ — G♭.
       От C♯ это G♯. Тональность G♯ мажор за пределами схемы.}
      {In D major, use D (I), G (IV, one step counterclockwise), and A (V,
       one step clockwise). Keep the interval spelling. A fifth above B
       is F♯, above C♭ it is G♭, and above C♯ it is G♯. The chart stops
       before \mbox{G♯ major}.}}%
    \GuideSection{\Localize{Аккорд на V ступени минора}{The chord on degree 5 in minor}}{\Localize
      {В E миноре V ступень — B. Натуральный минор даёт \mbox{Bm (v)}.
       Замените D на D♯, чтобы получить мажорный аккорд \mbox{B (V)}.}
      {In E minor, B is the root on degree 5. Natural minor gives Bm (v).
       Raise D to D♯ to get a B major triad (V).}}%
  };%
}
