-- Each row gives a semitone distance and its interval spellings:
-- abbreviation, diatonic number, English name, Russian name.
local intervals = {
  {0, {"P1", 1, "Perfect unison", "Чистая прима"}},
  {1, {"m2", 2, "Minor second", "Малая секунда"}},
  {2, {"M2", 2, "Major second", "Большая секунда"}},
  {3, {"m3", 3, "Minor third", "Малая терция"}},
  {4, {"M3", 3, "Major third", "Большая терция"}},
  {5, {"P4", 4, "Perfect fourth", "Чистая кварта"}},
  {6, {"A4", 4, "Augmented fourth", "Увеличенная кварта"},
      {"d5", 5, "Diminished fifth", "Уменьшённая квинта"}},
  {7, {"P5", 5, "Perfect fifth", "Чистая квинта"}},
  {8, {"m6", 6, "Minor sixth", "Малая секста"}},
  {9, {"M6", 6, "Major sixth", "Большая секста"}},
  {10, {"m7", 7, "Minor seventh", "Малая септима"}},
  {11, {"M7", 7, "Major seventh", "Большая септима"}},
  {12, {"P8", 8, "Perfect octave", "Чистая октава"}},
}

local openStrings = {64, 59, 55, 50, 45, 40} -- MIDI pitches, strings 1-6.
local letters = {"C", "D", "E", "F", "G", "A", "B"}
local naturalSteps = {0, 2, 4, 5, 7, 9, 11}

local function naturalLetter(pitch)
  for letter, semitones in ipairs(naturalSteps) do
    if semitones == pitch % 12 then
      return letter
    end
  end
  error("Expected a natural note for the diagram's starting pitch")
end

local function spellPitch(pitch, letter, octave)
  local naturalPitch = 12 * (octave + 1) + naturalSteps[letter]
  local alteration = pitch - naturalPitch
  local sign = alteration < 0 and "♭" or "♯"
  return letters[letter] .. string.rep(sign, math.abs(alteration)) .. octave
end

local function naturalName(pitch)
  return spellPitch(pitch, naturalLetter(pitch), pitch // 12 - 1)
end

local function intervalTarget(start, intervalNumber, distance)
  local targetLetter = naturalLetter(start) + intervalNumber - 1
  local targetOctave = start // 12 - 1 + (targetLetter - 1) // #letters
  return spellPitch(start + distance, (targetLetter - 1) % #letters + 1, targetOctave)
end

local function emit(command, ...)
  tex.sprint("\\" .. command .. "{" .. table.concat({...}, "}{") .. "}")
end

local function forEachDistance(start, printTarget)
  for _, row in ipairs(intervals) do
    local distance = row[1]
    local abbreviations, targets = {}, {}
    for index = 2, #row do
      local spelling = row[index]
      abbreviations[#abbreviations + 1] = spelling[1]
      targets[#targets + 1] = intervalTarget(start, spelling[2], distance)
    end
    printTarget(distance, table.concat(abbreviations, "/"), table.concat(targets, "/"))
  end
end

function PrintIntervalTable()
  local middleC, tableRow = 60, 0
  for _, row in ipairs(intervals) do
    local distance, firstRow = row[1], tableRow + 1
    for index = 2, #row do
      local spelling = row[index]
      tableRow = tableRow + 1
      emit("IntervalRow", tableRow, spelling[2], spelling[1], spelling[3], spelling[4],
        distance, intervalTarget(middleC, spelling[2], distance))
    end
    if tableRow > firstRow then
      emit("SharedDistanceFrame", firstRow, tableRow)
    end
  end
end

function PrintOneString()
  local startFret = 1
  local start = openStrings[2] + startFret
  forEachDistance(start, function(distance, abbreviation)
    emit("SingleTarget", startFret + distance, abbreviation)
  end)
end

function PrintAdjacentPairs()
  local startFret = 5
  for upper = 1, #openStrings - 1 do
    local lower = upper + 1
    local start = openStrings[lower] + startFret
    local openGap = openStrings[upper] - openStrings[lower]
    emit("PairGrid", upper, lower, naturalName(openStrings[upper]),
      naturalName(openStrings[lower]), startFret, naturalName(start), openGap)
    forEachDistance(start, function(distance, abbreviation, target)
      local targetFret = startFret + distance - openGap
      emit("PairTarget", upper, targetFret, abbreviation, target)
    end)
  end
end

function PrintGuitarTuning()
  local tuning = {}
  for stringNumber = #openStrings, 1, -1 do
    tuning[#tuning + 1] = naturalName(openStrings[stringNumber])
  end
  tex.sprint(table.concat(tuning, "-"))
end
\end{luacode*}
