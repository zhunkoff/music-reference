\newcommand{\ChartText}[6][base west]{%

\begin{luacode*}
local openStrings = {64, 59, 55, 50, 45, 40} -- MIDI pitches in string order 1-6.
local pitchNames = {
  {"C"}, {"C♯", "D♭"}, {"D"}, {"D♯", "E♭"}, {"E"}, {"F"},
  {"F♯", "G♭"}, {"G"}, {"G♯", "A♭"}, {"A"}, {"A♯", "B♭"}, {"B"},
}

local function spellPitch(pitch, showOctaves)
  local names = pitchNames[pitch % 12 + 1]
  local suffix = showOctaves and tostring(pitch // 12 - 1) or ""
  return names[1] .. suffix, names[2] and names[2] .. suffix or ""
end

function PrintFretboardTuning()
  local tuning = {}
  for stringNumber = #openStrings, 1, -1 do
    tuning[#tuning + 1] = spellPitch(openStrings[stringNumber], true)
  end
  tex.sprint(table.concat(tuning, "-"))
end

function PrintFretboardNotes(showOctaves)
  for stringNumber, openPitch in ipairs(openStrings) do
    for fret = 0, 12 do
      local first, second = spellPitch(openPitch + fret, showOctaves)
      tex.sprint(string.format([[\FretboardCell{%d}{%d}{%s}{%s}]],
        stringNumber, fret, first, second))
    end
  end
end
\end{luacode*}

\newcommand{\DrawFretboard}{%
  \fill[ChartHeader] (\ChartLeft,\StringTop) rectangle (\ChartRight,\HeaderTop);
  \foreach \row in {2,4,6} {%
    \fill[ChartStripe] (\ChartLeft,{\StringY{\row}})
      rectangle (\ChartRight,{\StringY{\row-1}});
  }%
  \fill[ChartOctave] ({\FretX{12}},\StringTop) rectangle (\ChartRight,\HeaderTop);
  \foreach \fret in {0,...,12} {%
    \draw[ChartRule,line width=.55bp] ({\FretX{\fret}},{\StringY{6}})
      -- ({\FretX{\fret}},\HeaderTop);
    \ChartText[base]{\FretCenter{\fret}}{\HeaderBaseline}{12}{ChartAccent}{\bfseries\fret}%
  }%
  \foreach \row in {0,...,5} {%
    \draw[ChartRule,line width=.55bp] (\ChartLeft,{\StringY{\row}})
      -- (\ChartRight,{\StringY{\row}});
  }%
  \draw[ChartRule,line width=.65bp] (\ChartLeft,{\StringY{6}})
    rectangle (\ChartRight,\HeaderTop);
  % The nut separates the open strings from fret 1. Outline the octave column.
  \draw[ChartAccent,line width=1.65bp] ({\FretX{1}},{\StringY{6}})
    -- ({\FretX{1}},\HeaderTop);
  \draw[ChartAccent,line width=1.25bp] ({\FretX{12}},{\StringY{6}})
    rectangle (\ChartRight,\HeaderTop);
  \draw[ChartMuted,line width=.55bp,shorten <=5bp,shorten >=5bp]
    (\ChartLeft,\HeaderTop) -- (\FretStart,\StringTop);
  \ChartText{\ChartLeft+5}{\StringTop+4.5}{9.5}{ChartAccent}
    {\bfseries\Localize{Струна}{String}}%
  \ChartText[base east]{\FretStart-5}{\HeaderTop-10}{9.5}{ChartAccent}
    {\bfseries\Localize{Лад}{Fret}}%
  \foreach \row in {1,...,6} {%
    \ChartText[base]{(\ChartLeft+\FretStart)/2}{\StringCenter{\row}-4}{12}{ChartAccent}
      {\bfseries\row}%
  }%
  \directlua{PrintFretboardNotes(\ifFretboardOctaves true\else false\fi)}%
}
