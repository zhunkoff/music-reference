% Chord name, first fret shown, six fret/finger pairs, barre spans.
% Pairs run from string 6 (low E) to string 1 (high E).
% Fret -1 means muted, 0 means open. Finger 0 draws no separate dot.
% A barre span is starting string / ending string / absolute fret.
% Example: 5/1/3 means a barre across strings 5 through 1 at fret 3.
\newcommand{\Chord}[4]{%
  \begin{scope}[shift={(138.9931+86.986225*\ChartColumnIndex,
                        467-69.5*\ChartRowIndex)}]
    \ChartText[base]{22.5}{11.3}{11.4}{ChartInk}{\bfseries#1}
    \foreach \fret/\finger [count=\stringindex from 0] in {#3} {
      \pgfmathsetmacro{\StringX}{9*\stringindex}
      \draw[ChartGrid,line width=.57bp] (\StringX,0) -- (\StringX,-38.8);
      \ifnum\fret=-1
        \draw[ChartInk,line width=.95bp]
          (\StringX-1.85,3.55) -- (\StringX+1.85,7.25)
          (\StringX-1.85,7.25) -- (\StringX+1.85,3.55);
      \else
        \ifnum\fret=0
          \draw[ChartInk,line width=.95bp] (\StringX,5.4) circle[radius=2.1bp];
        \fi
      \fi
    }
    \ifnum#2=1
      \draw[ChartInk,line width=1.65bp] (0,0) -- (45,0);
    \else
      \draw[ChartGrid,line width=.53bp] (0,0) -- (45,0);
      \ChartText[base east]{-7}{-7.2}{7.6}{ChartMuted}{\bfseries#2}
    \fi
    \foreach \fretline in {1,...,4} {
      \draw[ChartGrid,line width=.53bp] (0,-9.7*\fretline) -- (45,-9.7*\fretline);
    }
    \foreach \startstring/\endstring/\barrefret in {#4} {
      \draw[ChartInk,line width=6.6bp,line cap=round]
        ({9*(6-\startstring)},{-9.7*(\barrefret-#2+.5)}) --
        ({9*(6-\endstring)},{-9.7*(\barrefret-#2+.5)});
    }
    \foreach \fret/\finger [count=\stringindex from 0] in {#3} {
      \ifnum\finger>0
        \pgfmathsetmacro{\FingerX}{9*\stringindex}
        \pgfmathsetmacro{\FingerY}{-9.7*(\fret-#2+.5)}
        \fill[ChartInk] (\FingerX,\FingerY) circle[radius=3.85bp];
        \ifChartFingerNumbers
          \ChartText[base]{\FingerX}{\FingerY-6.8/3}
            {6.8}{white}{\bfseries\addfontfeatures{FakeBold=3}\finger}
        \fi
      \fi
    }
  \end{scope}
  \advance\ChartColumnIndex by 1
}
