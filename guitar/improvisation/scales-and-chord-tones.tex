local function chord_tones(key, chord)
  local root = chord[3]
  local third, fifth = (root + 1) % 7 + 1, (root + 3) % 7 + 1
  local third_steps = (key.intervals[third] - key.intervals[root]) % 12
  return {{root, "R"}, {third, third_steps == 3 and "♭3" or "3"}, {fifth, "5"}}
end

local function roles_by_note(key, chord)
  local roles = {}
  for _, tone in ipairs(chord_tones(key, chord)) do
    roles[key.scale[tone[1]]] = tone[2]
  end
  return roles
end

function PrintImprovisationScale(key_name)
  local key = keys[key_name]
  for degree, note in ipairs(key.scale) do
    tex.sprint(string.format([[\ScaleComparisonCell{%d}{%s}{%s}{%s}]],
      degree, key.degrees[degree], note, key.pentatonic[degree] and note or "—"))
  end
  tex.sprint(string.format([[\PositionLabel{%s}{%d}{%d}]], table.unpack(key.position)))
end

function PrintChordMaps(key_name)
  local key = keys[key_name]
  local _, first_fret, last_fret = table.unpack(key.position)
  local by_pitch = CagedMap.by_pitch(key.tonic, key.intervals, key.scale)
  local degree_by_pitch = CagedMap.by_pitch(key.tonic, key.intervals, key.degrees)
  for index, chord in ipairs(key.chords) do
    local roles = roles_by_note(key, chord)
    local members, present, missing = {}, {}, {}
    for _, tone in ipairs(chord_tones(key, chord)) do
      local degree, role = table.unpack(tone)
      local note = key.scale[degree]
      local membership = key.pentatonic[degree] and present or missing
      members[#members + 1] = role .. " = " .. note
      membership[#membership + 1] = note
    end
    tex.sprint(string.format([[\BeginChordMap{%d}{%s}{%s}{%s}{%s}{%s}{%d}{%d}]],
      index, chord[1], chord[2], table.concat(members, [[\quad ]]),
      table.concat(present, " · "), #missing > 0 and table.concat(missing, " · ") or "—",
      first_fret, last_fret))
    for string_number, string_data in ipairs(CagedMap.open_strings) do
      tex.sprint(string.format([[\MapString{%d}{%s}]], string_number, string_data[2]))
      for fret = first_fret, last_fret do
        local pitch = (string_data[1] + fret) % 12
        local note = by_pitch[pitch]
        if note then
          tex.sprint(string.format([[\MapNote{%d}{%d}{%s}{%s}{%s}]],
            string_number, fret, note, degree_by_pitch[pitch], roles[note] or ""))
        end
      end
    end
    tex.sprint([[\end{scope}]])
  end
end

local letter_index = {C=0, D=1, E=2, F=3, G=4, A=5, B=6}

function PrintPhrase(key_name)
  local key = keys[key_name]
  local by_pitch = CagedMap.by_pitch(key.tonic, key.intervals, key.scale)
  for bar_index, bar in ipairs(key.bars) do
    local chord = key.chords[bar[1]]
    local roles = roles_by_note(key, chord)
    tex.sprint(string.format([[\BeginBar{%d}{%s}]], bar_index, chord[1]))
    local onset, notes = 0, {}
    for index, event in ipairs(bar[2]) do
      local string_number, fret, duration, mark = table.unpack(event)
      local pitch = CagedMap.open_strings[string_number][1] + fret
      local note = by_pitch[pitch % 12]
      -- Written an octave above sounding pitch. Step 0 is E4, the bottom line.
      local step = 7 * (math.floor(pitch / 12) - 4) + letter_index[note] - 2
      notes[index] = {
        string_number = string_number, fret = fret, duration = duration,
        onset = onset, step = step, mark = mark, role = roles[note],
      }
      onset = onset + duration
    end
    for index, note in ipairs(notes) do
      local mark = ""
      if note.mark == "T" then
        mark = [[\TargetMark{]] .. note.role .. "}"
      elseif note.mark == "P" then
        mark = [[\PassingMark]]
      end
      tex.sprint(string.format([[\ScoreNote{%d}{%d}{%d}{%d}{%d}{%s}]],
        note.onset, note.step, note.duration, note.string_number, note.fret, mark))
      if note.duration > 1 then
        local direction = note.step < 4 and 1 or -1
        tex.sprint(string.format([[\ScoreStem{%d}{%d}{%d}{%d}]],
          note.onset, note.step, direction, 24 * direction))
      elseif note.onset % 2 == 0 then
        -- The eighth notes come in pairs on a beat, with a common stem direction.
        local next_note = notes[index + 1]
        local direction = note.step + next_note.step < 8 and 1 or -1
        local slope = math.max(-5, math.min(5, (next_note.step - note.step) * 1.5))
        local first_tip = 3.5 * note.step + 24 * direction
        local last_tip = first_tip + slope
        tex.sprint(string.format([[\ScoreBeam{%d}{%d}{%d}{%d}{%g}{%g}{%d}]],
          note.onset, note.step, next_note.onset, next_note.step, first_tip, last_tip, direction))
      end
    end
    tex.sprint([[\end{scope}]])
  end
end
