\begin{luacode*}
local openStrings = {64, 59, 55, 50, 45, 40} -- Sounding MIDI pitches, strings 1-6.

-- Rows are string groups 1-3 through 4-6. Columns are root position and
-- first and second inversions. Each triple runs from the bass string up.
-- Choose the lowest compact fretted position for each combination.
local voicings = {
  major = {
    {{5, 5, 3}, {9, 8, 8},    {12, 13, 12}},
    {{10, 9, 8}, {14, 12, 13}, {5, 5, 5}},
    {{15, 14, 12}, {7, 5, 5},  {10, 10, 9}},
    {{8, 7, 5}, {12, 10, 10},  {3, 3, 2}},
  },
  minor = {
    {{5, 4, 3}, {8, 8, 8},    {12, 13, 11}},
    {{10, 8, 8}, {13, 12, 13}, {5, 5, 4}},
    {{15, 13, 12}, {6, 5, 5},  {10, 10, 8}},
    {{8, 6, 5}, {11, 10, 10},  {3, 3, 1}},
  },
}

function PrintTriadVoicings(quality)
  local degrees = {[0] = "1", [7] = "5"}
  if quality == "major" then
    degrees[4] = "3"
  else
    degrees[3] = "♭3"
  end
  for row, group in ipairs(voicings[quality]) do
    local bassString = row + 2
    for column, frets in ipairs(group) do
      local firstFret = math.min(table.unpack(frets))
      local notes = {}
      for index, fret in ipairs(frets) do
        local stringNumber = bassString - index + 1
        local pitchClass = (openStrings[stringNumber] + fret) % 12
        notes[index] = string.format("%d/%d/%s", stringNumber, fret, degrees[pitchClass])
      end
      tex.sprint(string.format([[\DrawTriad{%d}{%d}{%d}{%d}{%s}]],
        row, column, firstFret, bassString, table.concat(notes, ",")))
    end
  end
end
\end{luacode*}

\newcommand{\FretWidth}{38}
\newcommand{\StringY}[1]{-24-9*(#1-1)}

% Draw a fretboard window whose first column is the lowest fret used.
% The string numbers and fret numbers reflect the full guitar, not the window.
\newcommand{\DrawTriad}[5]{%
  \begin{scope}[shift={({\ColumnLeft{#2}+51},{\RowTop{#1}})}]
    \foreach \fretOffset in {0,...,3} {%
      \pgfmathtruncatemacro{\fretNumber}{#3+\fretOffset}%
      \ChartText[base]{\FretWidth*(\fretOffset+.5)}{-14}{9.5}{ChartMuted}{\fretNumber}%
    }%
    \foreach \edge in {0,...,4} {%
      \draw[ChartRule,line width=.6bp] ({\FretWidth*\edge},{\StringY{1}})
        -- ({\FretWidth*\edge},{\StringY{6}});
    }%
    \foreach \stringNumber in {1,...,6} {%
      \pgfmathtruncatemacro{\played}{\stringNumber >= #4-2 && \stringNumber <= #4}%
      \ifnum\played=1\relax
        \draw[ChartMuted,line width=.48bp] (0,{\StringY{\stringNumber}})
          -- ({4*\FretWidth},{\StringY{\stringNumber}});
        \ChartText[center]{-25}{\StringY{\stringNumber}}{9}{ChartInk}{\stringNumber}%
      \else
        \draw[ChartRule,line width=.45bp] (0,{\StringY{\stringNumber}})
          -- ({4*\FretWidth},{\StringY{\stringNumber}});
        \ChartText[center]{-25}{\StringY{\stringNumber}}{9}{ChartMuted}{\stringNumber}%
        \ChartText[center]{-11}{\StringY{\stringNumber}}{9}{ChartInk}{×}%
      \fi
    }%
    \foreach \stringNumber/\fretNumber/\degree in {#5} {%
      \pgfmathsetmacro{\centerX}{\FretWidth*(\fretNumber-#3+.5)}%
      \ifdefstring{\degree}{1}{\def\DotColor{ChartAccent}}{\def\DotColor{ChartInk}}%
      \fill[\DotColor] ({\centerX-8},{\StringY{\stringNumber}}) circle[radius=2.9bp];
      \node[anchor=west,fill=white,inner xsep=1.1bp,inner ysep=0pt,
        text=ChartInk,font=\bfseries\fontsize{9bp}{9bp}\selectfont]
        at ({\centerX-2},{\StringY{\stringNumber}}) {\degree};
    }%
  \end{scope}%
}
