\begin{luacode*}
local major = {{0, 2, 4, 7, 9}, {"1", "2", "3", "5", "6"}}
local minor = {{0, 3, 5, 7, 10}, {"1", "♭3", "4", "5", "♭7"}}
local scales = {
  C = {0, major, {"C", "D", "E", "G", "A"}},
  A = {9, minor, {"A", "C", "D", "E", "G"}},
  D = {2, major, {"D", "E", "F♯", "A", "B"}},
  B = {11, minor, {"B", "D", "E", "F♯", "A"}},
}

function PrintScaleKey(key)
  local _, formula, names = table.unpack(scales[key])
  for index, note in ipairs(names) do
    tex.sprint(string.format([[\ScaleKeyCell{%d}{%s}{%s}]], index, formula[2][index], note))
  end
end

function PrintGridNotes(key, first_fret, last_fret)
  local tonic, formula = table.unpack(scales[key])
  local intervals, degrees = table.unpack(formula)
  CagedMap.print_grid_notes(CagedMap.by_pitch(tonic, intervals, degrees), first_fret, last_fret)
end

function PrintConnections()
  -- Both scales share these notes. On string 2, the last fret of each
  -- window holds a note also found in the next window, including C to A+12.
  local tonic, formula, names = table.unpack(scales.C)
  local names_by_pitch = CagedMap.by_pitch(tonic, formula[1], names)
  local open_b = CagedMap.open_strings[2][1]
  local positions = CagedMap.positions
  for index, position in ipairs(positions) do
    local name, _, fret = table.unpack(position)
    local next_position = positions[index + 1]
    local next_name = next_position and next_position[1] or positions[1][1] .. " (+12)"
    tex.sprint(string.format([[\SharedNote{%d}{%s}{%s}{%d}{%s}]],
      index, name, next_name, fret, names_by_pitch[(open_b + fret) % 12]))
  end
end
\end{luacode*}
