\begin{luacode*}
local major = {0, 2, 4, 5, 7, 9, 11}
local natural_minor = {0, 2, 3, 5, 7, 8, 10}
local scales = {
  C = {0, major, {"C", "D", "E", "F", "G", "A", "B"}},
  A = {9, natural_minor, {"A", "B", "C", "D", "E", "F", "G"}},
  D = {2, major, {"D", "E", "F♯", "G", "A", "B", "C♯"}},
  B = {11, natural_minor, {"B", "C♯", "D", "E", "F♯", "G", "A"}},
}

local function degree_label(intervals, degree)
  local alteration = intervals[degree] < major[degree] and "♭" or ""
  return alteration .. degree
end

function PrintScaleKey(key)
  local _, intervals, names = table.unpack(scales[key])
  for degree, note in ipairs(names) do
    tex.sprint(string.format([[\ScaleKeyCell{%d}{%s}{%s}]],
      degree, degree_label(intervals, degree), note))
  end
end

function PrintGridNotes(key, first_fret, last_fret)
  local tonic, intervals = table.unpack(scales[key])
  local degrees = {}
  for index in ipairs(intervals) do
    degrees[index] = degree_label(intervals, index)
  end
  CagedMap.print_grid_notes(CagedMap.by_pitch(tonic, intervals, degrees), first_fret, last_fret)
end
\end{luacode*}
