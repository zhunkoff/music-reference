  }

        \def\NEWPAX@link@type{GoTo}

      }
      { \book_newpax_uri: }
  }

\newcommand{\BookIncludeDocument}[3]{%
  \directlua{require("newpax").writenewpax("\luaescapestring{../#1}")}%
  \bookmark[dest=book-#2.1,level=0]{#3}%
  \newpaxsetup{destsuffix=#2}%
  % Record the page number on the first imported page for the next LaTeX pass.
  \includepdf[pages=-,fitpaper,link,linkname=book-#2,
    pagecommand*={\label{book-page-#2}}]{../#1.pdf}%
}
